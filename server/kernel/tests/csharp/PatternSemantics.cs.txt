public static class PatternRules
{
    public static int GuardedStatement(int value, int divisor)
    {
        switch (value)
        {
            case < 0: return -1;
            case 0 when 20 / divisor > 2: return 2;
            default: return 9;
        }
    }
    public static int GuardedExpression(int value, int divisor) => value switch
    {
        < 0 => -1,
        0 when 10 / divisor > 1 => 5,
        0 => 8,
        _ => 11
    };
    public static int Relational(int? value) => value switch
    {
        null => -1,
        >= 0 and < 10 => 1,
        not (< 0 or > 100) => 2,
        _ => 3
    };
    public static int BoundLocal(int? value)
    {
        switch (value)
        {
            case int number when number > 10: return number;
            case int number: return number * 2;
            default: return -1;
        }
    }
    public static int NonExhaustive(int value) => value switch { 1 => 10, 2 => 20 };
    public static int MutatingGuard(int value, int limit)
    {
        int result;
        switch (value)
        {
            case 0 when limit++ > 3: result = 100; break;
            case 0: result = 200; break;
            default: result = 300; break;
        }
        return result + limit;
    }
    public static bool IsSmall(int? value) => value is >= 0 and < 10;
}
