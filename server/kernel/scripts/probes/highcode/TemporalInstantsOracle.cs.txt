object Schema(Type type) {
    var nullable = Nullable.GetUnderlyingType(type);
    if (nullable != null) return new { kind = "nullable", value = Schema(nullable) };
    var kind = type == typeof(DateOnly) ? "date" : type == typeof(TimeOnly) ? "time" : type == typeof(TimeSpan) ? "duration" : type == typeof(DateTimeOffset) ? "offset" : type == typeof(int) ? "int" : type == typeof(long) ? "long" : type == typeof(bool) ? "bool" : type == typeof(string) ? "string" : "record";
    return kind != "record" ? (object)new { kind } : new { kind, fields = type.GetProperties().ToDictionary(p => p.Name, p => Schema(p.PropertyType)) };
}
object? Normalize(object? value) => value switch {
    null => null,
    DateOnly date => new { DayNumber = date.DayNumber },
    TimeOnly clock => new { Ticks = clock.Ticks.ToString(CultureInfo.InvariantCulture) },
    TimeSpan duration => new { Ticks = duration.Ticks.ToString(CultureInfo.InvariantCulture) },
    DateTimeOffset instant => new { UtcTicks = instant.UtcTicks.ToString(CultureInfo.InvariantCulture), OffsetMinutes = (int)instant.Offset.TotalMinutes },
    long number => number.ToString(CultureInfo.InvariantCulture),
    int or bool or string => value,
    _ => throw new Exception("Unexpected oracle result type: " + value.GetType())
};

    }
}
